----------------

    --db-url "postgresql://user:pass@host:port/database" \

# 6. Quick SQL verification

SELECT 'Accounts' as table_name, COUNT(*) FROM journey_accounts
UNION ALL SELECT 'Events', COUNT(*) FROM journey_events
UNION ALL SELECT 'KPIs', COUNT(*) FROM journey_kpis
UNION ALL SELECT 'Health', COUNT(*) FROM journey_health
UNION ALL SELECT 'Milestones', COUNT(*) FROM journey_milestones
